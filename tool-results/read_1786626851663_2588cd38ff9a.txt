    24→
    25→  const [token, setToken] = useState<string | null>(() => {
    26→    if (typeof window === 'undefined') return null
    27→    try { const s = localStorage.getItem('bandhan-admin-session'); return s ? JSON.parse(s).token : null } catch { return null }
    28→  })
    29→  const [email, setEmail] = useState('')
    30→  const [password, setPassword] = useState('')

    39→  const showToast = useCallback((msg: string) => { setToast(msg); setTimeout(() => setToast(''), 3000) }, [])

    45→      const r = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) })
    46→      const d = await r.json()
    47→      if (!r.ok) { setAuthError(d.error || d.message || sa.loginError); return }
    48→      if (d.data?.user?.roleName !== 'SUPER_ADMIN') { setAuthError(sa.notSuperAdmin); return }
    49→      setToken(d.data.token); setUserName(d.data.user?.profile?.displayName || d.data.user?.email || 'Admin')
    50→      localStorage.setItem('bandhan-admin-session', JSON.stringify(d.data))
    51→    } catch { setAuthError('Network error') } finally { setLoading(false) }
    52→  }, [email, password, sa])
    53→
    54→  const api = useCallback(async (url: string, opts?: RequestInit) => {
    55→    if (!token) return null
    56→    const r = await fetch(url, { ...opts, headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}`, ...(opts?.headers as Record<string, string>) } })
    57→    const d = await r.json()
    58→    if (r.status === 401 || r.status === 403) { setToken(null); localStorage.removeItem('bandhan-admin-session'); return null }
    59→    return d

    65→    const d = await api('/api/v1/admin/ops/dashboard')
    66→    if (d?.data) setDash(d.data)
    67→    setDashLoading(false)
    68→  }, [token, api])
    69→

    71→  useEffect(() => { if (token && tab === 'dashboard') fetchDash() }, [tab, token, fetchDash])
    72→
    73→  const logout = () => { setToken(null); setDash(null); setUserName(''); localStorage.removeItem('bandhan-admin-session'); setTab('dashboard') }

    80→    { id: 'offers', icon: Gift, labelKey: sa.offers },
    81→    { id: 'membership', icon: Crown, labelKey: sa.membership },
    82→    { id: 'settings', icon: Settings, labelKey: sa.settings },
    83→  ]
    84→
    85→  // ─── LOGIN ───
    86→  if (!token) return (
    87→    <div className="flex min-h-screen flex-col bg-ivory">
    88→      <header className="sticky top-0 z-40 border-b border-border/40 bg-white/90 backdrop-blur-xl">
    89→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-3 px-4 sm:px-6 md:h-16">
    90→          <button onClick={onBack} className="flex h-9 w-9 items-center justify-center rounded-xl transition-all hover:bg-cream"><ArrowLeft className="h-5 w-5 text-foreground" /></button>
    91→          <BandhanLogo /><SahuSamajFlag size="sm" className="hidden sm:inline-flex" />
    92→        </div>
    93→      </header>
    94→      <main className="flex flex-1 items-center justify-center px-4 py-12">
    95→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.5 }} className="w-full max-w-sm rounded-2xl border border-border/40 bg-white p-6 shadow-sm sm:p-8">
    96→          <div className="mb-6 flex justify-center"><div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-burgundy/10"><ShieldCheck className="h-6 w-6 text-burgundy" /></div></div>
    97→          <h1 className="text-center text-xl font-bold text-foreground font-serif">{sa.loginTitle}</h1>
    98→          <p className="mt-1 text-center text-xs text-muted-foreground">{sa.loginSubtitle}</p>
    99→          {authError && <div className="mt-4 rounded-lg bg-red-50 px-3 py-2 text-xs font-medium text-red-600">{authError}</div>}
   100→          <div className="mt-5 space-y-3">
   101→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.email}</label><input type="email" value={email} onChange={e => setEmail(e.target.value)} onKeyDown={e => e.key === 'Enter' && !loading && handleLogin()} placeholder="admin@bandhan.com" className="h-11 w-full rounded-xl border border-border/60 bg-white px-4 text-sm text-foreground placeholder:text-muted-foreground/60 transition-all focus:border-burgundy/30 focus:outline-none focus:ring-2 focus:ring-burgundy/10" /></div>
   102→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.password}</label><input type="password" value={password} onChange={e => setPassword(e.target.value)} onKeyDown={e => e.key === 'Enter' && !loading && handleLogin()} placeholder="............" className="h-11 w-full rounded-xl border border-border/60 bg-white px-4 text-sm text-foreground placeholder:text-muted-foreground/60 transition-all focus:border-burgundy/30 focus:outline-none focus:ring-2 focus:ring-burgundy/10" /></div>

   190→/* ═══════════ USERS TAB ═══════════ */
   191→function UsersTab({ sa, api, showToast }: { sa: Record<string, string>; api: (u: string, o?: RequestInit) => Promise<any>; showToast: (m: string) => void }) {
   192→  const [search, setSearch] = useState('')
   193→  const [users, setUsers] = useState<any[]>([])
   194→  const [loading, setLoading] = useState(false)
   195→  const [selected, setSelected] = useState<string[]>([])
   196→
   197→  const fetchUsers = useCallback(async () => {
   198→    setLoading(true)
   199→    const d = await api(`/api/v1/admin/ops/users?limit=50${search ? `&search=${encodeURIComponent(search)}` : ''}`)
   200→    if (d?.data?.users) setUsers(d.data.users)
   201→    setLoading(false)
   202→  }, [api, search])
   203→
   204→  useEffect(() => { fetchUsers() }, [fetchUsers])
   205→
   206→  const toggleSel = (id: string) => setSelected(p => p.includes(id) ? p.filter(x => x !== id) : [...p, id])
   207→  const toggleAll = () => setSelected(users.length === selected.length ? [] : users.map((u: any) => u.id))
   208→
   209→  const bulkAction = async (action: string, extra?: Record<string, string>) => {
   210→    if (!selected.length) return
   211→    const d = await api('/api/v1/admin/ops/users', { method: 'POST', body: JSON.stringify({ action, userIds: selected, ...extra }) })
   212→    if (d?.data) { showToast(`${action} done for ${d.data.processed} users`); setSelected([]); fetchUsers() }
   213→    else if (d?.error) showToast(d.error)
   214→  }
   215→
   216→  return (
   217→    <div>
   218→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   219→        <h1 className="text-lg font-bold text-foreground font-serif">{sa.users}</h1>
   220→        {selected.length > 0 && (
   221→          <div className="flex items-center gap-2">
   222→            <span className="text-xs text-muted-foreground">{selected.length} {sa.selected}</span>
   223→            <button onClick={() => bulkAction('APPROVE')} className="btn-bandhan-primary py-1 px-3 text-xs">{sa.approve}</button>
   224→            <button onClick={() => bulkAction('BLOCK')} className="rounded-lg bg-red-50 px-3 py-1 text-xs font-medium text-red-600">{sa.block}</button>
   225→            <button onClick={() => bulkAction('UNBLOCK')} className="rounded-lg bg-emerald-50 px-3 py-1 text-xs font-medium text-emerald-600">{sa.unblock}</button>
   226→          </div>
   227→        )}
   228→      </div>
   229→      <div className="relative mb-4"><Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground/60" /><input value={search} onChange={e => setSearch(e.target.value)} placeholder={sa.searchUsers} className="h-11 w-full rounded-xl border border-border/60 bg-white pl-10 pr-4 text-sm" /></div>
   230→      {loading ? <div className="py-12 text-center text-sm text-muted-foreground">{sa.loading}</div> : users.length === 0 ? <p className="py-12 text-center text-sm text-muted-foreground">{sa.noUsers}</p> : (
   231→        <div className="overflow-x-auto rounded-xl border border-border/40 bg-white">
   232→          <table className="w-full text-left text-xs">
   233→            <thead className="border-b border-border/30 bg-ivory/60"><tr>
   234→              <th className="p-3"><input type="checkbox" checked={users.length === selected.length && users.length > 0} onChange={toggleAll} /></th>
   235→              <th className="p-3 font-medium text-muted-foreground">Email</th><th className="hidden p-3 font-medium text-muted-foreground sm:table-cell">Role</th><th className="p-3 font-medium text-muted-foreground">Status</th><th className="hidden p-3 font-medium text-muted-foreground md:table-cell">Tier</th><th className="hidden p-3 font-medium text-muted-foreground lg:table-cell">Joined</th>
   236→            </tr></thead>
   237→            <tbody>{users.map((u: any) => (
   238→              <tr key={u.id} className={`border-b border-border/20 last:border-0 ${selected.includes(u.id) ? 'bg-burgundy/5' : ''}`} onClick={() => toggleSel(u.id)}>
   239→                <td className="p-3"><input type="checkbox" checked={selected.includes(u.id)} readOnly /></td>
   240→                <td className="p-3 font-medium text-foreground">{u.email || u.phone || '-'}</td>
   241→                <td className="hidden p-3 text-muted-foreground sm:table-cell">{u.role?.name || '-'}</td>
   242→                <td className="p-3"><span className={`inline-block rounded-full px-2 py-0.5 text-[0.65rem] font-medium ${u.status === 'ACTIVE' ? 'bg-emerald-50 text-emerald-600' : u.status === 'BLOCKED' ? 'bg-red-50 text-red-600' : 'bg-amber-50 text-amber-600'}`}>{u.status}</span></td>
   243→                <td className="hidden p-3 text-muted-foreground md:table-cell">{u.membershipTier || '-'}</td>
   244→                <td className="hidden p-3 text-muted-foreground lg:table-cell">{u.createdAt ? new Date(u.createdAt).toLocaleDateString() : '-'}</td>
   245→              </tr>
   246→            ))}</tbody>
   247→          </table>
   248→        </div>
   249→      )}
   250→    </div>
   251→  )
   252→}

   254→/* ═══════════ ADS TAB ═══════════ */
   255→function AdsTab({ sa, api, showToast }: { sa: Record<string, string>; api: (u: string, o?: RequestInit) => Promise<any>; showToast: (m: string) => void }) {
   256→  const [ads, setAds] = useState<any[]>([])
   257→  const [rates, setRates] = useState<Record<string, string>>({ AD_HERO_RATE: '5000', AD_MIDDLE_RATE: '3000', AD_BOTTOM_RATE: '2000' })
   258→  const [showForm, setShowForm] = useState(false)
   259→  const [form, setForm] = useState({ name: '', module: 'GENERAL', placement: 'HERO', mediaType: 'IMAGE', mediaUrl: '', linkUrl: '', ctaText: '', ctaTextHi: '', language: 'ALL', device: 'ALL', priority: 0, startDate: '', endDate: '' })
   260→
   261→  const fetchAds = useCallback(async () => {
   262→    const d = await api('/api/v1/admin/ops/advertisements?limit=50')
   263→    if (d?.data?.campaigns) setAds(d.data.campaigns)
   264→  }, [api])
   265→
   266→  const fetchRates = useCallback(async () => {
   267→    const d = await api('/api/v1/admin/ops/settings?category=ADVERTISEMENT')
   268→    if (d?.data?.settings) {
   269→      const r: Record<string, string> = {}
   270→      d.data.settings.forEach((s: any) => { if (s.key.startsWith('AD_') && s.key.endsWith('_RATE')) r[s.key] = s.value })
   271→      if (Object.keys(r).length > 0) setRates(r)
   272→    }
   273→  }, [api])
   274→
   275→  useEffect(() => { fetchAds(); fetchRates() }, [fetchAds, fetchRates])
   276→
   277→  const createAd = async () => {

   486→function SettingsTab({ sa, api, token, showToast }: { sa: Record<string, string>; api: (u: string, o?: RequestInit) => Promise<any>; token: string; showToast: (m: string) => void }) {
   487→  const [section, setSection] = useState('password' as string)
   488→  const [curPwd, setCurPwd] = useState('')
   489→  const [newPwd, setNewPwd] = useState('')
   490→  const [cfmPwd, setCfmPwd] = useState('')
   491→  const [pwdLoading, setPwdLoading] = useState(false)
   492→  const [pwdErr, setPwdErr] = useState('')
   493→  const [siteCfg, setSiteCfg] = useState({ SITE_NAME: 'Bandhan', SITE_DESCRIPTION: 'Sahu Samaj Connects', CONTACT_EMAIL: '', CONTACT_PHONE: '', MAINTENANCE_MODE: 'false' })
   494→  const [secCfg, setSecCfg] = useState({ MAX_LOGIN_ATTEMPTS: '5', SESSION_DURATION_DAYS: '30' })
   495→  const [settingsLoading, setSettingsLoading] = useState(false)
   496→
   497→  const fetchSettings = useCallback(async () => {
   498→    const d = await api('/api/v1/admin/ops/settings')
   499→    if (d?.data?.settings) {
   500→      const g: Record<string, string> = {}, s: Record<string, string> = {}
   501→      d.data.settings.forEach((st: any) => { const target = st.category === 'GENERAL' ? g : st.category === 'SECURITY' ? s : null; if (target) target[st.key] = st.value })
   502→      if (Object.keys(g).length) setSiteCfg(p => ({ ...p, ...g }))
   503→      if (Object.keys(s).length) setSecCfg(p => ({ ...p, ...s }))
   504→    }
   505→  }, [api])
   506→  useEffect(() => { fetchSettings() }, [fetchSettings])
   507→
   508→  const changePassword = async () => {
   509→    if (newPwd !== cfmPwd) { setPwdErr(sa.passwordMismatch); return }
   510→    if (newPwd.length < 8 || !/[A-Z]/.test(newPwd) || !/[0-9]/.test(newPwd)) { setPwdErr(sa.passwordError); return }
   511→    setPwdErr(''); setPwdLoading(true)
   512→    const r = await fetch('/api/v1/auth/password', { method: 'PUT', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ currentPassword: curPwd, newPassword: newPwd }) })
   513→    const d = await r.json()
   514→    if (d.success) { showToast(sa.passwordChanged); setCurPwd(''); setNewPwd(''); setCfmPwd('') }
   515→    else setPwdErr(d.error || 'Failed')
   516→    setPwdLoading(false)
   517→  }
   518→
   519→  const saveSiteSettings = async () => {
   520→    setSettingsLoading(true)
   521→    const settings = Object.entries(siteCfg).map(([key, value]) => ({ key, value, category: 'GENERAL', description: key, isPublic: true }))
   522→    const d = await api('/api/v1/admin/settings', { method: 'PUT', body: JSON.stringify({ settings }) })
   523→    if (d?.data) showToast(sa.settingsSaved)
   524→    setSettingsLoading(false)
   525→  }
   526→
   527→  const saveSecSettings = async () => {
   528→    setSettingsLoading(true)
   529→    const settings = Object.entries(secCfg).map(([key, value]) => ({ key, value, category: 'SECURITY', description: key }))
   530→    const d = await api('/api/v1/admin/ops/settings', { method: 'PUT', body: JSON.stringify({ settings }) })
   531→    if (d?.data) showToast(sa.settingsSaved)
   532→    setSettingsLoading(false)
   533→  }
   534→
   535→  return (
   536→    <div>
