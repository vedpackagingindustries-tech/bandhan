    24→
    25→  const [token, setToken] = useState<string | null>(() => {
    26→    if (typeof window === 'undefined') return null
    27→    try { const s = localStorage.getItem('bandhan-admin-session'); return s ? JSON.parse(s).token : null } catch { return null }
    28→  })
    29→  const [email, setEmail] = useState('')

    33→  const [tab, setTab] = useState<TabId>('dashboard')
    34→  const [dash, setDash] = useState<Record<string, Record<string, number>> | null>(null)

    39→  const showToast = useCallback((msg: string) => { setToast(msg); setTimeout(() => setToast(''), 3000) }, [])

    45→      const r = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) })
    46→      const d = await r.json()
    47→      if (!r.ok) { setAuthError(d.error || d.message || sa.loginError); return }
    48→      if (d.data?.user?.roleName !== 'SUPER_ADMIN') { setAuthError(sa.notSuperAdmin); return }
    49→      setToken(d.data.token); setUserName(d.data.user?.profile?.displayName || d.data.user?.email || 'Admin')
    50→      localStorage.setItem('bandhan-admin-session', JSON.stringify(d.data))
    51→    } catch { setAuthError('Network error') } finally { setLoading(false) }
    52→  }, [email, password, sa])
    53→
    54→  const api = useCallback(async (url: string, opts?: RequestInit) => {
    55→    if (!token) return null
    56→    const r = await fetch(url, { ...opts, headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}`, ...(opts?.headers as Record<string, string>) } })
    57→    const d = await r.json()
    58→    if (r.status === 401 || r.status === 403) { setToken(null); localStorage.removeItem('bandhan-admin-session'); return null }

    69→

    72→
    73→  const logout = () => { setToken(null); setDash(null); setUserName(''); localStorage.removeItem('bandhan-admin-session'); setTab('dashboard') }

    82→    { id: 'settings', icon: Settings, labelKey: sa.settings },

    85→  // ─── LOGIN ───

    87→    <div className="flex min-h-screen flex-col bg-ivory">
    88→      <header className="sticky top-0 z-40 border-b border-border/40 bg-white/90 backdrop-blur-xl">
    89→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-3 px-4 sm:px-6 md:h-16">
    90→          <button onClick={onBack} className="flex h-9 w-9 items-center justify-center rounded-xl transition-all hover:bg-cream"><ArrowLeft className="h-5 w-5 text-foreground" /></button>
    91→          <BandhanLogo /><SahuSamajFlag size="sm" className="hidden sm:inline-flex" />
    92→        </div>
    93→      </header>
    94→      <main className="flex flex-1 items-center justify-center px-4 py-12">
    95→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.5 }} className="w-full max-w-sm rounded-2xl border border-border/40 bg-white p-6 shadow-sm sm:p-8">
    96→          <div className="mb-6 flex justify-center"><div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-burgundy/10"><ShieldCheck className="h-6 w-6 text-burgundy" /></div></div>
    97→          <h1 className="text-center text-xl font-bold text-foreground font-serif">{sa.loginTitle}</h1>
    98→          <p className="mt-1 text-center text-xs text-muted-foreground">{sa.loginSubtitle}</p>
    99→          {authError && <div className="mt-4 rounded-lg bg-red-50 px-3 py-2 text-xs font-medium text-red-600">{authError}</div>}
   100→          <div className="mt-5 space-y-3">
   101→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.email}</label><input type="email" value={email} onChange={e => setEmail(e.target.value)} onKeyDown={e => e.key === 'Enter' && !loading && handleLogin()} placeholder="admin@bandhan.com" className="h-11 w-full rounded-xl border border-border/60 bg-white px-4 text-sm text-foreground placeholder:text-muted-foreground/60 transition-all focus:border-burgundy/30 focus:outline-none focus:ring-2 focus:ring-burgundy/10" /></div>

   190→/* ═══════════ USERS TAB ═══════════ */
   191→function UsersTab({ sa, api, showToast }: { sa: Record<string, string>; api: (u: string, o?: RequestInit) => Promise<any>; showToast: (m: string) => void }) {
   192→  const [search, setSearch] = useState('')
   193→  const [users, setUsers] = useState<any[]>([])
   194→  const [loading, setLoading] = useState(false)
   195→  const [selected, setSelected] = useState<string[]>([])
   196→
   197→  const fetchUsers = useCallback(async () => {
   198→    setLoading(true)
   199→    const d = await api(`/api/v1/admin/ops/users?limit=50${search ? `&search=${encodeURIComponent(search)}` : ''}`)
   200→    if (d?.data?.users) setUsers(d.data.users)
   201→    setLoading(false)
   202→  }, [api, search])
   203→
   204→  useEffect(() => {
   205→    let active = true
   206→    api(`/api/v1/admin/ops/users?limit=50${search ? `&search=${encodeURIComponent(search)}` : ''}`)
   207→      .then(d => { if (active && d?.data?.users) setUsers(d.data.users) })
   208→      .catch(() => {})
   209→    return () => { active = false }
   210→  }, [api, search])
   211→
   212→  const toggleSel = (id: string) => setSelected(p => p.includes(id) ? p.filter(x => x !== id) : [...p, id])
   213→  const toggleAll = () => setSelected(users.length === selected.length ? [] : users.map((u: any) => u.id))
   214→
   215→  const bulkAction = async (action: string, extra?: Record<string, string>) => {
   216→    if (!selected.length) return
   217→    const d = await api('/api/v1/admin/ops/users', { method: 'POST', body: JSON.stringify({ action, userIds: selected, ...extra }) })
   218→    if (d?.data) { showToast(`${action} done for ${d.data.processed} users`); setSelected([]); fetchUsers() }
   219→    else if (d?.error) showToast(d.error)
   220→  }
   221→
   222→  return (
   223→    <div>
   224→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   225→        <h1 className="text-lg font-bold text-foreground font-serif">{sa.users}</h1>
   226→        {selected.length > 0 && (
   227→          <div className="flex items-center gap-2">
   228→            <span className="text-xs text-muted-foreground">{selected.length} {sa.selected}</span>
   229→            <button onClick={() => bulkAction('APPROVE')} className="btn-bandhan-primary py-1 px-3 text-xs">{sa.approve}</button>
   230→            <button onClick={() => bulkAction('BLOCK')} className="rounded-lg bg-red-50 px-3 py-1 text-xs font-medium text-red-600">{sa.block}</button>
   231→            <button onClick={() => bulkAction('UNBLOCK')} className="rounded-lg bg-emerald-50 px-3 py-1 text-xs font-medium text-emerald-600">{sa.unblock}</button>
   232→          </div>
   233→        )}
   234→      </div>
   235→      <div className="relative mb-4"><Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground/60" /><input value={search} onChange={e => setSearch(e.target.value)} placeholder={sa.searchUsers} className="h-11 w-full rounded-xl border border-border/60 bg-white pl-10 pr-4 text-sm" /></div>
   236→      {loading ? <div className="py-12 text-center text-sm text-muted-foreground">{sa.loading}</div> : users.length === 0 ? <p className="py-12 text-center text-sm text-muted-foreground">{sa.noUsers}</p> : (
   237→        <div className="overflow-x-auto rounded-xl border border-border/40 bg-white">
   238→          <table className="w-full text-left text-xs">
   239→            <thead className="border-b border-border/30 bg-ivory/60"><tr>
   240→              <th className="p-3"><input type="checkbox" checked={users.length === selected.length && users.length > 0} onChange={toggleAll} /></th>
   241→              <th className="p-3 font-medium text-muted-foreground">Email</th><th className="hidden p-3 font-medium text-muted-foreground sm:table-cell">Role</th><th className="p-3 font-medium text-muted-foreground">Status</th><th className="hidden p-3 font-medium text-muted-foreground md:table-cell">Tier</th><th className="hidden p-3 font-medium text-muted-foreground lg:table-cell">Joined</th>
   242→            </tr></thead>
   243→            <tbody>{users.map((u: any) => (
   244→              <tr key={u.id} className={`border-b border-border/20 last:border-0 ${selected.includes(u.id) ? 'bg-burgundy/5' : ''}`} onClick={() => toggleSel(u.id)}>
   245→                <td className="p-3"><input type="checkbox" checked={selected.includes(u.id)} readOnly /></td>
   246→                <td className="p-3 font-medium text-foreground">{u.email || u.phone || '-'}</td>
   247→                <td className="hidden p-3 text-muted-foreground sm:table-cell">{u.role?.name || '-'}</td>
   248→                <td className="p-3"><span className={`inline-block rounded-full px-2 py-0.5 text-[0.65rem] font-medium ${u.status === 'ACTIVE' ? 'bg-emerald-50 text-emerald-600' : u.status === 'BLOCKED' ? 'bg-red-50 text-red-600' : 'bg-amber-50 text-amber-600'}`}>{u.status}</span></td>
   249→                <td className="hidden p-3 text-muted-foreground md:table-cell">{u.membershipTier || '-'}</td>
   250→                <td className="hidden p-3 text-muted-foreground lg:table-cell">{u.createdAt ? new Date(u.createdAt).toLocaleDateString() : '-'}</td>
   251→              </tr>
   252→            ))}</tbody>
   253→          </table>
   254→        </div>
   255→      )}
   256→    </div>
   257→  )
   258→}

   261→function AdsTab({ sa, api, showToast }: { sa: Record<string, string>; api: (u: string, o?: RequestInit) => Promise<any>; showToast: (m: string) => void }) {
   262→  const [ads, setAds] = useState<any[]>([])
   263→  const [rates, setRates] = useState<Record<string, string>>({ AD_HERO_RATE: '5000', AD_MIDDLE_RATE: '3000', AD_BOTTOM_RATE: '2000' })
   264→  const [showForm, setShowForm] = useState(false)
   265→  const [form, setForm] = useState({ name: '', module: 'GENERAL', placement: 'HERO', mediaType: 'IMAGE', mediaUrl: '', linkUrl: '', ctaText: '', ctaTextHi: '', language: 'ALL', device: 'ALL', priority: 0, startDate: '', endDate: '' })
   266→
   267→  const fetchAds = useCallback(async () => {
   268→    const d = await api('/api/v1/admin/ops/advertisements?limit=50')
   269→    if (d?.data?.campaigns) setAds(d.data.campaigns)
   270→  }, [api])
   271→
   272→  const fetchRates = useCallback(async () => {
   273→    const d = await api('/api/v1/admin/ops/settings?category=ADVERTISEMENT')
   274→    if (d?.data?.settings) {
   275→      const r: Record<string, string> = {}
   276→      d.data.settings.forEach((s: any) => { if (s.key.startsWith('AD_') && s.key.endsWith('_RATE')) r[s.key] = s.value })
   277→      if (Object.keys(r).length > 0) setRates(r)
   278→    }
   279→  }, [api])
   280→
   281→  useEffect(() => {

   555→    if (newPwd !== cfmPwd) { setPwdErr(sa.passwordMismatch); return }
   556→    if (newPwd.length < 8 || !/[A-Z]/.test(newPwd) || !/[0-9]/.test(newPwd)) { setPwdErr(sa.passwordError); return }
   557→    setPwdErr(''); setPwdLoading(true)
   558→    const r = await fetch('/api/v1/auth/password', { method: 'PUT', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ currentPassword: curPwd, newPassword: newPwd }) })
   559→    const d = await r.json()
   560→    if (d.success) { showToast(sa.passwordChanged); setCurPwd(''); setNewPwd(''); setCfmPwd('') }
   561→    else setPwdErr(d.error || 'Failed')
   562→    setPwdLoading(false)
   563→  }
   564→
   565→  const saveSiteSettings = async () => {
   566→    setSettingsLoading(true)
   567→    const settings = Object.entries(siteCfg).map(([key, value]) => ({ key, value, category: 'GENERAL', description: key, isPublic: true }))
   568→    const d = await api('/api/v1/admin/settings', { method: 'PUT', body: JSON.stringify({ settings }) })
   569→    if (d?.data) showToast(sa.settingsSaved)
   570→    setSettingsLoading(false)
   571→  }
   572→
   573→  const saveSecSettings = async () => {
   574→    setSettingsLoading(true)
   575→    const settings = Object.entries(secCfg).map(([key, value]) => ({ key, value, category: 'SECURITY', description: key }))
   576→    const d = await api('/api/v1/admin/ops/settings', { method: 'PUT', body: JSON.stringify({ settings }) })
   577→    if (d?.data) showToast(sa.settingsSaved)
   578→    setSettingsLoading(false)
   579→  }
   580→
   581→  return (
   582→    <div>
   583→      <div className="mb-4 flex gap-2">
   584→        {[['password', sa.changePassword], ['general', sa.siteSettings], ['secSettings', sa.secSettings]].map(([k, l]) => <button key={k} onClick={() => setSection(k)} className={`rounded-lg px-3 py-1.5 text-xs font-medium transition-all ${section === k ? 'bg-burgundy text-white' : 'border border-border/40 text-muted-foreground hover:bg-cream'}`}>{l}</button>)}
   585→      </div>
   586→      {section === 'password' && (
   587→        <div className="mx-auto max-w-sm rounded-xl border border-border/40 bg-white p-5">
   588→          {pwdErr && <div className="mb-3 rounded-lg bg-red-50 px-3 py-2 text-xs font-medium text-red-600">{pwdErr}</div>}
   589→          <div className="space-y-3">
   590→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.currentPassword}</label><input type="password" value={curPwd} onChange={e => setCurPwd(e.target.value)} className="h-11 w-full rounded-xl border border-border/60 px-4 text-sm" /></div>
   591→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.newPassword}</label><input type="password" value={newPwd} onChange={e => setNewPwd(e.target.value)} className="h-11 w-full rounded-xl border border-border/60 px-4 text-sm" /></div>
   592→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.confirmPassword}</label><input type="password" value={cfmPwd} onChange={e => setCfmPwd(e.target.value)} onKeyDown={e => e.key === 'Enter' && changePassword()} className="h-11 w-full rounded-xl border border-border/60 px-4 text-sm" /></div>
   593→          </div>
   594→          <button onClick={changePassword} disabled={pwdLoading} className="btn-bandhan-primary mt-4 w-full py-2 text-sm disabled:opacity-50">{pwdLoading ? sa.saving : sa.save}</button>
   595→        </div>
   596→      )}
   597→      {section === 'general' && (
   598→        <div className="rounded-xl border border-border/40 bg-white p-5">
   599→          <h3 className="mb-3 text-sm font-bold text-foreground">{sa.siteSettings}</h3>
   600→          <div className="space-y-3">
   601→            <Inp label={sa.siteName} value={siteCfg.SITE_NAME || ''} onChange={v => setSiteCfg(p => ({ ...p, SITE_NAME: v }))} />
   602→            <div><label className="mb-1 block text-xs font-medium text-foreground">{sa.siteDescription}</label><textarea value={siteCfg.SITE_DESCRIPTION || ''} onChange={e => setSiteCfg(p => ({ ...p, SITE_DESCRIPTION: e.target.value }))} rows={2} className="w-full rounded-xl border border-border/60 px-4 py-2 text-sm" /></div>
   603→            <Inp label={sa.contactEmail} value={siteCfg.CONTACT_EMAIL || ''} onChange={v => setSiteCfg(p => ({ ...p, CONTACT_EMAIL: v }))} />
   604→            <Inp label={sa.contactPhone} value={siteCfg.CONTACT_PHONE || ''} onChange={v => setSiteCfg(p => ({ ...p, CONTACT_PHONE: v }))} />
   605→          </div>
   606→          <button onClick={saveSiteSettings} disabled={settingsLoading} className="btn-bandhan-primary mt-4 py-2 px-6 text-sm disabled:opacity-50">{settingsLoading ? sa.saving : sa.save}</button>
   607→        </div>
   608→      )}
   609→      {section === 'secSettings' && (
   610→        <div className="rounded-xl border border-border/40 bg-white p-5">
   611→          <h3 className="mb-3 text-sm font-bold text-foreground">{sa.security}</h3>
   612→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   613→            <Inp label={sa.maxLoginAttempts} value={secCfg.MAX_LOGIN_ATTEMPTS || ''} onChange={v => setSecCfg(p => ({ ...p, MAX_LOGIN_ATTEMPTS: v }))} />
   614→            <Inp label={sa.sessionDuration} value={secCfg.SESSION_DURATION_DAYS || ''} onChange={v => setSecCfg(p => ({ ...p, SESSION_DURATION_DAYS: v }))} />
   615→          </div>
   616→          <button onClick={saveSecSettings} disabled={settingsLoading} className="btn-bandhan-primary mt-4 py-2 px-6 text-sm disabled:opacity-50">{settingsLoading ? sa.saving : sa.save}</button>
   617→        </div>
   618→      )}
   619→    </div>
   620→  )
   621→}

   633→function M({ l, v, g, w, b }: { l: string; v: number | string; g?: boolean; w?: boolean; b?: boolean }) {

   642→}
